{{- $src := .Get "src" -}} {{- with .Page.Resources.GetMatch $src -}} {{- $img := . -}} {{- if eq $img.MediaType.SubType "svg" -}} {{- /* SVG: vector, no processing, no width/height */ -}} {{- else -}} {{- $isPNGorJPG := (or (eq $img.MediaType.SubType "png") (eq $img.MediaType.SubType "jpeg")) -}} {{- if and $isPNGorJPG (ge $img.Width 640) -}} {{- $small := ($img.Resize "640x webp q82").RelPermalink -}} {{- $medium := ($img.Resize "1024x webp q82").RelPermalink -}} {{- $large := ($img.Resize "1600x webp q82").RelPermalink -}} {{- else -}} {{- end -}} {{- end -}} {{- else -}} {{- errorf "figure shortcode: image %q not found in %s" $src $.Page.File.Path -}} {{- end -}}
